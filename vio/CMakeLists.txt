cmake_minimum_required(VERSION 3.20)
project(rover_vio CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
option(ROVER_NATIVE "Optimize for this machine (-march=native)" ON)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  add_compile_options(-Wall -Wextra -Wno-unused-parameter)
  if(ROVER_NATIVE)
    add_compile_options(-march=native)
  endif()
endif()
# MinGW: link the C++ runtime statically (another libstdc++ DLL on PATH crashes at run time)
if(MINGW)
  add_link_options(-static)
endif()

add_library(rover STATIC src/estimator.cpp src/frontend.cpp src/features.cpp src/image.cpp src/euroc.cpp src/loop.cpp)
target_include_directories(rover PUBLIC include)
find_package(Threads)
if(Threads_FOUND)
  target_link_libraries(rover PUBLIC Threads::Threads)
endif()

add_executable(rover-vio tools/run_euroc.cpp)
add_executable(rover-stream tools/stream.cpp)
add_executable(rover-voc tools/train_voc.cpp)
foreach(t rover-vio rover-stream rover-voc)
  target_link_libraries(${t} PRIVATE rover)
endforeach()

enable_testing()
foreach(t test_core test_eigen test_png)
  add_executable(${t} tests/${t}.cpp)
  target_link_libraries(${t} PRIVATE rover)
  add_test(NAME ${t} COMMAND ${t})
endforeach()
